### 优化1: 改进Few-shot示例选择

```python
# data_processor.py

def generate_few_shot_examples_v2(self, examples_per_category: int = 5):
    """
    改进的Few-shot示例生成（v2.0）
    """
    few_shot_examples = {}
    
    for cat, records in categories.items():
        # 策略1: 选择多样化的示例（不只是最长的）
        examples = []
        
        # 按描述长度分组
        short = [r for r in records if len(r['事件描述']) < 50]
        medium = [r for r in records if 50 <= len(r['事件描述']) < 150]
        long = [r for r in records if len(r['事件描述']) >= 150]
        
        # 从各组选择，保证多样性
        examples.extend(long[:2])
        examples.extend(medium[:2])
        examples.extend(short[:1])
        
        few_shot_examples[cat] = examples[:examples_per_category]
    
    return few_shot_examples
```

### 优化2: 添加对比学习示例

```python
# qwen_classifier.py

def _build_prompt_with_contrast(self, event_description, event_type, ...):
    """
    添加对比学习的提示词构建
    """
    prompt = f"""
    当前事件类型：{event_type}
    
    可选分类：{valid_categories}
    
    分类示例：
    {few_shot_examples}
    
    ⚠️ 注意区分（对于易混淆的分类）:
    """
    
    # 添加对比说明
    if event_type == "城市管理":
        prompt += """
        - 街面秩序: 关注通行和秩序（占道、违停、流动摊贩）
        - 市容环境: 关注环境和卫生（垃圾、绿化、清洁）
        """
    
    if event_type == "消防安全":
        prompt += """
        - 电气线路问题: 线路本身问题（私拉乱接、老化）
        - 电动车违规充电: 充电行为问题（室内充电、飞线）
        """
    
    return prompt
```
